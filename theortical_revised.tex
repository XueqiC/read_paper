\section{Theoretical Analysis}
\label{sec:theory}

\rev{This section gives a local account of three phenomena that ReAD acts on, namely spillover, diminishing returns, and noisy reward estimates. Appendix~\ref{sec:app_theory} gives the assumptions and proofs.}

\rev{\textbf{Local cross-capability transfer.} Let $\theta_t$ be the student
parameters after decision step $t$ and $\Delta\theta_t=\theta_t-\theta_{t-1}$.
Assume that (i) the expected update under allocation $\mathbf{w}_t$ is a
mixture $\sum_{c}w_{t,c}\,d_c(\theta_{t-1})$ of capability-conditioned update
directions, and (ii) each measured capability is locally a differentiable
readout $\tilde{\mathbf{s}}(\theta)$. Then, by Proposition~\ref{prop:transfer}, the
change of each capability $c$ at step $t$ decomposes as}
\begin{equation}
\Delta s_{t,c}
\approx
\sum_{c'\in\mathcal{C}} w_{t,c'}\,\Gamma_{c,c'}(\theta_{t-1}),
\qquad
\Gamma_{c,c'}(\theta)
:=
\nabla_\theta[\tilde{\mathbf{s}}(\theta)]_c^\top d_{c'}(\theta).
\label{eq:transfer_decomp_short}
\end{equation}
\rev{$\Gamma$ is a local analogue of the empirical transfer matrix. An off-diagonal entry is nonzero whenever the update direction for $c'$ is not orthogonal to the readout direction of $c$. Shared representations likely make this common, so budget given to one capability can alter others at every step.}

\rev{\textbf{Diminishing returns and budget waste.} Let $b_c\ge0$ be the tokens
assigned to capability $c$, $\sum_c b_c\le B$, and suppose task-aligned gains
are concave in the budget,
$\mathbb{E}[U_\tau(S)-U_\tau(S_0)]\approx\sum_c r_{\tau,c}\,G_{\tau,c}(b_c)$
with $G_{\tau,c}$ nondecreasing and concave. An optimal allocation equalizes
weighted marginal gains across active capabilities (Proposition~\ref{prop:kkt}),
so moving a small budget from a single target $\bar c$ to another capability
$c'$ helps whenever
$r_{\tau,c'}G'_{\tau,c'}(0)>r_{\tau,\bar c}G'_{\tau,\bar c}(b_{\bar c})$. Once the target saturates, further tokens can be worth less than tokens spent elsewhere (Observation~2). This proxy ignores spillover, which ReAD's reward in
Eq.~\eqref{eq:reward} penalizes separately.}

\rev{\textbf{Uncertainty-aware allocation.} Let
$\overline{R}(\mathbf{x},\mathbf{w})$ be the conditional expected proxy reward
of allocation $\mathbf{w}$ in context $\mathbf{x}$. If the ensemble is
calibrated, $|\mu-\overline{R}|\le\kappa\,\sigma$ with high probability, the
action chosen by Eq.~\eqref{eq:ucb} is within $2\kappa\,\sigma(\mathbf{x}_t,\mathbf{w}_t)$
of the best action in expected reward (Proposition~\ref{prop:regret}). Large errors can then occur only where the allocator's own uncertainty is large, which is also where the bonus directs exploration. The statement holds locally at each step and offers no global guarantee.}
